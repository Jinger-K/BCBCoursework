wc -l ../data/fasta/407228326.fasta ../data/fasta/407228412.fasta ../data/fasta/e_coli.fasta

# This command is used to count newline characters in FASTA files. The count for the two numbered files is one lower than the number of lines displayed in a text editor, as they do not end with newline.

#2
tail -n +2 ../data/fasta/e_coli.fasta

# This command displays all lines from the second line of e_coli.fasta. The header is excluded because it contains descriptive information rather than sequence data. In "tail -n +2", "tail" prints the final part of a file, "-n" specifies the output in terms of lines, and "+2" means starting from the second line.

#3
tail -n +2 ../data/fasta/e_coli.fasta | xargs printf '%s' | wc -c

# This command excludes the header, joins the sequence lines without adding spaces or newlines, and counts the resulting bytes. The supplied sequence uses ASCII symbols, so each byte represents one sequence position, including ambiguous nucleotide symbols.

#4
tail -n +2 ../data/fasta/e_coli.fasta | xargs printf '%s' | grep -o 'ATGC' | wc -l

# This command excludes the header and joins the sequence lines so that motifs spanning line breaks can be found. "grep -o" prints each ATGC match on a separate line, and "wc -l" counts those matches rather than the number of input lines containing the motifs.

#5
echo "$(tail -n +2 ../data/fasta/e_coli.fasta | grep -o '[AT]' | wc -l) / $(tail -n +2 ../data/fasta/e_coli.fasta | grep -o '[CG]' | wc -l)" | bc -l

# This command separately counts the total number of A and T and the total number of C and G without ambiguous positions. The counts are divided using "bc -l" to calculate (A+T)/(C+G) with decimal precision, rather than using integer division. The supplied sequence has a nonzero GC count. If that count were zero, the ratio would be undefined.
